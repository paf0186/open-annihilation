cmake_minimum_required(VERSION 3.24)
project(oa_sim_ground_orders LANGUAGES CXX)
include(CTest)
add_library(oa-sim-ground-orders src/orders.cpp src/goals.cpp src/navigator.cpp src/runtime.cpp src/movement_map.cpp src/search_trace.cpp src/search_heap.cpp src/search_worker.cpp)
add_library(oa::sim::ground_orders ALIAS oa-sim-ground-orders)
target_include_directories(oa-sim-ground-orders PUBLIC include)
target_link_libraries(oa-sim-ground-orders PUBLIC oa-data-match-rules oa-sim-simulation-state oa-sim-unit-spawn oa-sim-spatial-state oa-data-unit-definitions oa-base-game-math)
target_link_libraries(oa-sim-ground-orders PUBLIC oa-sim-unit-movement oa-base-game-math oa-sim-unit-spawn
  PRIVATE oa-base-bytes)
target_compile_features(oa-sim-ground-orders PUBLIC cxx_std_20)
if(TARGET oa-options)
  target_link_libraries(oa-sim-ground-orders PUBLIC oa-options)
endif()
if(BUILD_TESTING)
  add_executable(oa-ground-orders-test tests/orders_test.cpp)
  target_link_libraries(oa-ground-orders-test PRIVATE oa-sim-ground-orders)
  add_test(NAME ground-orders COMMAND oa-ground-orders-test)
  add_executable(oa-search-heap-test tests/search_heap_test.cpp)
  target_link_libraries(oa-search-heap-test PRIVATE oa-sim-ground-orders oa-test-support)
  add_test(NAME search-heap COMMAND oa-search-heap-test)
  add_executable(oa-search-worker-test tests/search_worker_test.cpp)
  target_link_libraries(oa-search-worker-test PRIVATE oa-sim-ground-orders oa-data-limits)
  add_test(NAME search-worker COMMAND oa-search-worker-test)
  add_executable(oa-movement-map-test tests/movement_map_test.cpp)
  target_link_libraries(oa-movement-map-test PRIVATE oa-sim-ground-orders)
  add_test(NAME movement-map COMMAND oa-movement-map-test)
endif()
